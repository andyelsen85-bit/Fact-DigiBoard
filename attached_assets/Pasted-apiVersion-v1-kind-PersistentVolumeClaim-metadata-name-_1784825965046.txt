apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: digiboard-postgres-pvc
  namespace: default
spec:
  accessModes:
    - ReadWriteOnce
  resources:
    requests:
      storage: 5Gi
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: digiboard-postgres
  namespace: default
spec:
  replicas: 1
  selector:
    matchLabels:
      app: digiboard-postgres
  template:
    metadata:
      labels:
        app: digiboard-postgres
    spec:
      containers:
      - name: postgres
        image: postgres:16-alpine
        env:
        - name: POSTGRES_USER
          value: "digiboard"
        - name: POSTGRES_PASSWORD
          value: "digiboard_secret"
        - name: POSTGRES_DB
          value: "digiboard"
        # Behebt den lost+found Fehler in containerd v3 vorab:
        - name: PGDATA
          value: "/var/lib/postgresql/data/pgdata"
        ports:
        - containerPort: 5432
        volumeMounts:
        - name: postgres-data
          mountPath: /var/lib/postgresql/data
      volumes:
      - name: postgres-data
        persistentVolumeClaim:
          claimName: digiboard-postgres-pvc
---
apiVersion: v1
kind: Service
metadata:
  name: digiboard-postgres
  namespace: default
spec:
  ports:
  - port: 5432
    targetPort: 5432
  selector:
    app: digiboard-postgres
---
